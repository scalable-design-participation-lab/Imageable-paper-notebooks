Vegetation + Brick percentages

One of the observations we make in the Results section is that more facades dominated by brick and vegetation were observed in Boston. As evidence we present the percentage of buildings dominated for each material per-city. We compute those in this notebook.

feature_names = validation_instances_boston[0]["feature_names"]
print(feature_names)

MATERIALS = ['material_pct_concrete', 'material_pct_metal', 'material_pct_glass', 'material_pct_plastic', 'material_pct_asphalt', 'material_pct_brick', 'material_pct_human_body', 'material_pct_wood', 'material_pct_plaster', 'material_pct_leaf']

SCALER_PATH = "/Users/legariapena.j/Documents/imageable-2026/imageable/data/models/cluster_ensemble_scaler.pkl"

import joblib
scaler = joblib.load(SCALER_PATH)

indices_materials = [feature_names.index(x) for x in MATERIALS]
def material_percentages(instances):
    features = np.array([x["features"] for x in instances])
    return scaler.inverse_transform(features)[:, indices_materials]


boston_instances_materials = material_percentages(validation_instances_boston)
monterrey_instances_materials = material_percentages(validation_instances_monterrey)
