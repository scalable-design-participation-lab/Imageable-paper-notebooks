Auxiliary functions

Following the structure of the first experiment let’s first define some auxiliary functions.

import numpy as np
import joblib

def get_rmse(
        y_pred,
        y_real
    ):
    rmse = 0
    n = len(y_pred)
    for i in range(0,n):
        value_pred = y_pred[i]
        value_real = y_real[i]

        rmse +=(1/n)*(value_pred - value_real)**2
    
    rmse = np.sqrt(rmse)
    return rmse


def get_mae(y_pred, y_real):
    mae = 0
    n = len(y_pred)
    for i in range(0,n):
        value_pred = y_pred[i]
        value_real = y_real[i]
        mae += (1/n)*np.abs(value_pred - value_real)
    return mae

def get_mean_signed_error(y_pred, y_real):
    n = len(y_pred)
    diffs = []
    for i in range(0,n):
        value_pred = y_pred[i]
        value_real = y_real[i]
        diffs.append(value_pred - value_real)

    return np.mean(diffs)

def filter_data_by_indices(data, indices):
    new_data = []
    for i in range(0,len(data)):
        if(i in indices):
            new_data.append(data[i])
    
    return new_data


def get_cluster_indices(data, cluster_value, cluster_key = "cluster"):
    n = len(data)
    indices = []
    for i in range(0,n):
        data_point = data[i]
        if(data_point.get(cluster_key, None) == cluster_value):
            indices.append(i)
        
    return indices

Function that applies the experiment

In this experiment we model performances for each building archetype (cluster) using two error metrics: \(RMSE\) and \(MAE\).

The calculations are simple, we

  1. Divide the validation instances depending on the cluster that they belong to.

  2. For each cluster we obtain the building height estimation error with the single XGBoost model and the cluster-then-predict system that we propose in this work.

  3. We get

\[\begin{equation} \Delta = RMSE_{\text{CTP}} - RMSE_{\text{single}} \end{equation}\]

class ExperimentResults:

    def __init__(
        self,
        sample_size,
        mae_single,
        rmse_single,
        signed_single,
        mae_ctp,
        rmse_ctp,
        signed_ctp,
        improvement_rmse,
        improvement_mae
    ):

        self.mae_single = mae_single
        self.rmse_single = rmse_single
        self.signed_single = signed_single
        self.mae_ctp = mae_ctp
        self.rmse_ctp = rmse_ctp
        self.signed_ctp = signed_ctp
        self.improvement_mae = improvement_mae
        self.improvement_rmse = improvement_rmse
        self.sample_size = sample_size
    

    def to_dict(self):
        output_dictionary = {}
        output_dictionary["sample_size"] = self.sample_size
        output_dictionary["rmse_single"] = self.rmse_single
        output_dictionary["mae_single"] = self.mae_single
        output_dictionary["signed_single"] = self.signed_single
        output_dictionary["rmse_ctp"] = self.rmse_ctp
        output_dictionary["mae_ctp"] = self.mae_ctp
        output_dictionary["signed_ctp"] = self.signed_ctp
        output_dictionary["improvement_mae"] = self.improvement_mae
        output_dictionary["improvement_rmse"] = self.improvement_rmse

        return output_dictionary




def main_experiment_2(
    cluster_vectors,
    model_single,
    model_ctp,
    real_heights
):

    sample_size = len(cluster_vectors)
    predictions_single = model_single.predict(cluster_vectors)
    predictions_ctp = model_ctp.predict(cluster_vectors)

    #Single XGBoost results
    mae_single = get_mae(predictions_single, real_heights)
    rmse_single = get_rmse(predictions_single, real_heights)
    signed_single = get_mean_signed_error(predictions_single, real_heights)

    #Cluster-then-predict results
    mae_ctp = get_mae(predictions_ctp, real_heights)
    rmse_ctp = get_rmse(predictions_ctp, real_heights)
    signed_ctp = get_mean_signed_error(predictions_ctp, real_heights)

    #Error improvements
    improvement_mae = mae_single - mae_ctp
    improvement_rmse = rmse_single - rmse_ctp

    result = ExperimentResults(
        sample_size = sample_size,
        mae_single = mae_single, 
        rmse_single = rmse_single,
        signed_single = signed_single,
        mae_ctp = mae_ctp, 
        rmse_ctp = rmse_ctp, 
        signed_ctp = signed_ctp, 
        improvement_mae = improvement_mae, 
        improvement_rmse = improvement_rmse
    )

    return result
