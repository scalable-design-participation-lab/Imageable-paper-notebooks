Differentiating between training and validation

We open the train/val data file, which contains information about which instances were selected for validation and which for training per city.

import json

DICTIONARIES_PATH = "/Users/legariapena.j/Documents/imageable-2026/imageable/data/training_validation_data/data.json"

with open(DICTIONARIES_PATH) as f:
    data = json.load(f)

#IDs Monterrey
ids_monterrey_data = [x["building_id"] for x in data if x["city"] == "Monterrey"]
#IDs Boston
ids_boston_data = [x["building_id"] for x in data if x["city"] == "Boston"]
#Data types (train vs. val) Monterrey
data_type_monterrey = {x["building_id"]: x["data_type"] for x in data if x["city"] == "Monterrey"}
#Data types (train vs. val) Boston
data_type_boston = {x["building_id"]: x["data_type"] for x in data if x["city"] == "Boston"}

gdf_monterrey_data = gdf_monterrey[gdf_monterrey["building_id"].isin(ids_monterrey_data)]
gdf_boston_data = gdf_boston[gdf_boston["building_id"].isin(ids_boston_data)]

#Now we tag according to the data type
gdf_monterrey_data["data_type"] = [data_type_monterrey[b] for b in gdf_monterrey_data["building_id"]]
gdf_boston_data["data_type"] = [data_type_boston[b] for b in gdf_boston_data["building_id"]]

Plot Monterrey

Let’s use display_footprints_categorical to display the training and validation buildings for Monterrey.

#We can now add all buildings that are not in the training/validation set as unlabelled data
ground_truth_monterrey = ground_truth_monterrey.to_crs(gdf_monterrey_data.crs)
ground_truth_monterrey["data_type"] = "Not in either"
monterrey_labels = ground_truth_monterrey["building_id"].map(data_type_monterrey)
ground_truth_monterrey.loc[monterrey_labels.notna(), "data_type"] = monterrey_labels.dropna()

from imageable.visualization.footprints_2d import display_footprints_categorical

import os

CARTO_API_KEY = os.environ.get("CARTO_API_KEY")

DRACULA_PINK = "#ff79c6"
DRACULA_CYAN = "#8BE9FD"
DRACULA_COMMENT = "#6272a4"

plot_mty = ground_truth_monterrey.copy()
plot_mty["data_type"] = pd.Categorical(
    plot_mty["data_type"],
    categories=["train", "validation", "Not in either"],
    ordered=True,
)
plot_mty = plot_mty.sort_values("data_type")

n_train = (plot_mty["data_type"] == "train").sum()
n_val = (plot_mty["data_type"] == "validation").sum()
n_other = (plot_mty["data_type"] == "Not in either").sum()

display_footprints_categorical(
    gdf=plot_mty,
    column="data_type",
    colors=[DRACULA_PINK, DRACULA_CYAN, DRACULA_COMMENT],
    base_gdf=all_buildings_monterrey,
    title="Monterrey",
    subtitle=f"Training: {n_train:,}    Validation: {n_val:,}    Discarded: {n_other:,}",
    title_y=1.16,
    subtitle_y=1.05,
    annotate_counts=False,
    focus_on_gdf_bounds=True,
    add_basemap=True,
